"use client";
import { useState, useRef, useEffect } from "react";
import { jsPDF } from "jspdf";
import html2canvas from "html2canvas";
import { useAuth } from "@/context/AuthContext";

// TimetablePDF Component (updated to match dashboard style)
const TimetablePDF = ({ data, name, darkMode }) => {
  const tableRef = useRef(null);

  const handleDownload = async () => {
    try {
      const tableClone = tableRef.current.cloneNode(true);
      document.body.appendChild(tableClone);

      const doc = new jsPDF("p", "mm", "a4");
      const canvas = await html2canvas(tableClone, {
        scale: 2,
        logging: false,
        useCORS: true,
        backgroundColor: darkMode ? "#1a202c" : "#ffffff",
      });

      const imgData = canvas.toDataURL("image/png");
      const imgWidth = doc.internal.pageSize.getWidth() - 20;
      const imgHeight = (canvas.height * imgWidth) / canvas.width;

      doc.addImage(imgData, "PNG", 10, 10, imgWidth, imgHeight);
      doc.save("smata_timetable.pdf");
      document.body.removeChild(tableClone);
    } catch (error) {
      console.error("Error generating PDF:", error);
    }
  };

  return (
    <div className="space-y-4">
      <div>
        <h1 className="text-2xl font-bold">Smata</h1>
      </div>
      <div className="flex justify-between items-center">
        <h2 className="text-xl font-semibold">Study Timetable</h2>
        <button
          onClick={handleDownload}
          className={`px-4 py-2 rounded-lg ${
            darkMode
              ? "bg-blue-600 hover:bg-blue-700"
              : "bg-blue-500 hover:bg-blue-600"
          } text-white`}
        >
          Download PDF
        </button>
      </div>

      <div className="overflow-x-auto">
        <table
          ref={tableRef}
          className={`w-full rounded-lg overflow-hidden ${
            darkMode ? "bg-gray-800 text-gray-100" : "bg-white text-gray-800"
          }`}
        >
          <thead>
            <tr className={darkMode ? "bg-gray-700" : "bg-gray-100"}>
              <th className="p-3 text-left">Day</th>
              <th className="p-3 text-left">Time</th>
              <th className="p-3 text-left">Subject</th>
              <th className="p-3 text-left">Duration</th>
            </tr>
          </thead>
          <tbody>
            {data.map((session, index) => (
              <tr
                key={index}
                className={`border-t ${
                  darkMode ? "border-gray-700" : "border-gray-200"
                }`}
              >
                <td className="p-3">{session.day}</td>
                <td className="p-3">{session.timeSlot}</td>
                <td className="p-3">{session.course}</td>
                <td className="p-3">{session.time}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
};

  const [adCountdown, setAdCountdown] = useState(15);
  const [adWatched, setAdWatched] = useState(false);

  const userName = user?.displayName || "User";

  // Color palette for subjects
  const colors = [
    "#3498db",
    "#e74c3c",
    "#2ecc71",
    "#f39c12",
    "#9b59b6",
    "#1abc9c",
    "#d35400",
    "#34495e",
  ];

  // Current time and date
  const [currentTime, setCurrentTime] = useState("");
  const [currentDate, setCurrentDate] = useState("");

  useEffect(() => {
    const updateDateTime = () => {
      const now = new Date();
      setCurrentTime(
        now.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })
      );
      setCurrentDate(
        now.toLocaleDateString([], {
          month: "short",
          day: "numeric",
          year: "numeric",
        })
      );
    };

    updateDateTime();
    const interval = setInterval(updateDateTime, 60000);
    return () => clearInterval(interval);

    let timer;
    if (showAd && adCountdown > 0) {

      setAdWatched(true);

              </div>
            )}

            {/* Calendar Section */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">Calendar</h2>
              <p className="opacity-70">No items found</p>
            </div>
          </div>

          {/* Right Column */}
          <div className="space-y-6">
            {/* Activities */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">Activities</h2>
              <h3 className="font-medium mb-2">Upcoming exams</h3>
              <p className="opacity-70 mb-4">No exams scheduled yet</p>
              <button className="flex items-center text-blue-500 font-medium">
                <span className="mr-1">+</span> Quick add task
              </button>
            </div>

            {/* Focus Timer */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">Focus Timer</h2>
              <div className="mb-3">
                <label className="block font-medium mb-1">Subject</label>
                <input
                  type="text"
                  className={`w-full p-3 rounded-lg border ${borderColor} ${inputBg}`}
                  placeholder="What are you studying?"
                />
              </div>
              <div className="mb-4">
                <label className="block font-medium mb-1">Time</label>
                <input
                  type="time"
                  className={`w-full p-3 rounded-lg border ${borderColor} ${inputBg}`}
                />
              </div>
              <button
                className={`w-full py-3 px-4 rounded-lg font-medium ${
                  darkMode
                    ? "bg-green-600 hover:bg-green-700"
                    : "bg-green-500 hover:bg-green-600"
                } text-white`}
              >
                Start Focus Session
              </button>
            </div>

            {/* Date Navigation */}
            <div
              className={`p-4 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <div className="flex justify-between">
                {["Today", "Tomorrow", "Custom Date"].map((tab) => (
                  <button
                    key={tab}
                    onClick={() => setActiveTab(tab)}
                    className={`px-3 py-2 rounded-md font-medium ${
                      activeTab === tab
                        ? darkMode
                          ? "bg-blue-900 text-blue-300"
                          : "bg-blue-100 text-blue-600"
                        : darkMode
                        ? "hover:bg-gray-700"
                        : "hover:bg-gray-100"
                    }`}
                  >
                    {tab}
                  </button>
                ))}
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};

export default Dashboard;
