

"use client";
import { useState, useRef, useEffect } from "react";
import { jsPDF } from "jspdf";
import html2canvas from "html2canvas";
import { useAuth } from "@/context/AuthContext";

// TimetablePDF Component (updated to match dashboard style)
const TimetablePDF = ({ data, name, darkMode }) => {
  const tableRef = useRef(null);

  const handleDownload = async () => {
    try {
      const tableClone = tableRef.current.cloneNode(true);
      document.body.appendChild(tableClone);

      const doc = new jsPDF("p", "mm", "a4");
      const canvas = await html2canvas(tableClone, {
        scale: 2,
        logging: false,
        useCORS: true,
        backgroundColor: darkMode ? "#1a202c" : "#ffffff",
      });

      const imgData = canvas.toDataURL("image/png");
      const imgWidth = doc.internal.pageSize.getWidth() - 20;
      const imgHeight = (canvas.height * imgWidth) / canvas.width;

      doc.addImage(imgData, "PNG", 10, 10, imgWidth, imgHeight);
      doc.save("smata_timetable.pdf");
      document.body.removeChild(tableClone);
    } catch (error) {
      console.error("Error generating PDF:", error);
    }
  };

  return (
    <div className="space-y-4">
      <div>
        <h1 className="text-2xl font-bold">Smata</h1>
      </div>
      <div className="flex justify-between items-center">
        <h2 className="text-xl font-semibold">Study Timetable</h2>
        <button
          onClick={handleDownload}
          className={`px-4 py-2 rounded-lg ${
            darkMode
              ? "bg-blue-600 hover:bg-blue-700"
              : "bg-blue-500 hover:bg-blue-600"
          } text-white`}
        >
          Download PDF
        </button>
      </div>

      <div className="overflow-x-auto">
        <table
          ref={tableRef}
          className={`w-full rounded-lg overflow-hidden ${
            darkMode ? "bg-gray-800 text-gray-100" : "bg-white text-gray-800"
          }`}
        >
          <thead>
            <tr className={darkMode ? "bg-gray-700" : "bg-gray-100"}>
              <th className="p-3 text-left">Day</th>
              <th className="p-3 text-left">Time</th>
              <th className="p-3 text-left">Subject</th>
              <th className="p-3 text-left">Duration</th>
            </tr>
          </thead>
          <tbody>
            {data.map((session, index) => (
              <tr
                key={index}
                className={`border-t ${
                  darkMode ? "border-gray-700" : "border-gray-200"
                }`}
              >
                <td className="p-3">{session.day}</td>
                <td className="p-3">{session.timeSlot}</td>
                <td className="p-3">{session.course}</td>
                <td className="p-3">{session.time}</td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
};

// Main Dashboard Component
const Dashboard = () => {
  const { user, signOutUser } = useAuth();
  const [darkMode, setDarkMode] = useState(false);
  const [name, setName] = useState("");
  const [subjects, setSubjects] = useState([]);
  const [table, setTable] = useState([]);
  const [activeTab, setActiveTab] = useState("Today");
  const [isGenerating, setIsGenerating] = useState(false);
  const [generationProgress, setGenerationProgress] = useState(0);
  const [showAd, setShowAd] = useState(false);
  const [adCountdown, setAdCountdown] = useState(15);
  const [adWatched, setAdWatched] = useState(false);

  // Safe user display name
  const userName = user?.displayName || "User";

  // Color palette for subjects
  const colors = [
    "#3498db",
    "#e74c3c",
    "#2ecc71",
    "#f39c12",
    "#9b59b6",
    "#1abc9c",
    "#d35400",
    "#34495e",
  ];

  // Current time and date
  const [currentTime, setCurrentTime] = useState("");
  const [currentDate, setCurrentDate] = useState("");

  useEffect(() => {
    const updateDateTime = () => {
      const now = new Date();
      setCurrentTime(
        now.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })
      );
      setCurrentDate(
        now.toLocaleDateString([], {
          month: "short",
          day: "numeric",
          year: "numeric",
        })
      );
    };

    updateDateTime();
    const interval = setInterval(updateDateTime, 60000);
    return () => clearInterval(interval);
  }, []);

  // Ad countdown timer
  useEffect(() => {
    let timer;
    if (showAd && adCountdown > 0) {
      timer = setTimeout(() => {
        setAdCountdown(adCountdown - 1);
      }, 1000);
    } else if (showAd && adCountdown === 0) {
      setAdWatched(true);
    }
    return () => clearTimeout(timer);
  }, [showAd, adCountdown]);

  // Generation progress simulation
  useEffect(() => {
    let timer;
    if (isGenerating && generationProgress < 100) {
      timer = setTimeout(() => {
        setGenerationProgress(generationProgress + 100 / 15);
      }, 1000);
    }
    return () => clearTimeout(timer);
  }, [isGenerating, generationProgress]);

  const toggleDarkMode = () => {
    setDarkMode(!darkMode);
    document.documentElement.classList.toggle("dark", !darkMode);
  };

  const handleGenerateTimetable = () => {
    setIsGenerating(true);
    setGenerationProgress(0);
    setShowAd(true);

    // Simulate timetable generation
    setTimeout(() => {
      const mockTimetable = [
        {
          day: "Monday",
          timeSlot: "08:00 - 10:00",
          course: "Mathematics",
          time: "2h",
        },
        {
          day: "Tuesday",
          timeSlot: "10:00 - 12:00",
          course: "Physics",
          time: "2h",
        },
        {
          day: "Wednesday",
          timeSlot: "14:00 - 16:00",
          course: "Chemistry",
          time: "2h",
        },
      ];
      setTable(mockTimetable);
      setIsGenerating(false);
    }, 15000);
  };

  const handleContinueAfterAd = () => {
    setShowAd(false);
  };

  // Styling classes based on dark mode
  const bgColor = darkMode ? "bg-gray-900" : "bg-gray-50";
  const textColor = darkMode ? "text-gray-100" : "text-gray-800";
  const cardBg = darkMode ? "bg-gray-800" : "bg-white";
  const borderColor = darkMode ? "border-gray-700" : "border-gray-200";
  const inputBg = darkMode ? "bg-gray-700" : "bg-white";

  return (
    <div
      className={`min-h-screen ${bgColor} ${textColor} transition-colors duration-300`}
    >
      {/* Ad Overlay */}
      {showAd && (
        <div className="fixed inset-0 bg-black bg-opacity-80 z-50 flex items-center justify-center p-4">
          <div className={`max-w-md w-full rounded-xl p-6 ${cardBg}`}>
            <h2 className="text-xl font-semibold mb-4">Advertisement</h2>
            <p className="mb-4">
              Please watch this short ad to generate your timetable
            </p>

            <div className="w-full h-48 bg-gray-600 rounded-lg flex items-center justify-center mb-4">
              <p className="text-white">
                {adWatched
                  ? "Thank you!"
                  : `Sponsored Content (${adCountdown}s)`}
              </p>
            </div>

            <div className="flex justify-end">
              {adWatched ? (
                <button
                  onClick={handleContinueAfterAd}
                  className={`px-4 py-2 rounded-lg ${
                    darkMode
                      ? "bg-green-600 hover:bg-green-700"
                      : "bg-green-500 hover:bg-green-600"
                  } text-white`}
                >
                  Continue
                </button>
              ) : (
                <button
                  className={`px-4 py-2 rounded-lg ${
                    darkMode ? "bg-gray-700" : "bg-gray-200"
                  }`}
                  disabled
                >
                  Please wait...
                </button>
              )}
            </div>
          </div>
        </div>
      )}

      <div className="container mx-auto p-4 max-w-6xl">
        {/* Header */}
        <header className="flex justify-between items-center mb-6">
          <div>
            <h1 className="text-2xl font-bold">Smata</h1>
            <p className="text-sm opacity-70">
              {currentTime} · {currentDate}
            </p>
          </div>
          <div className="min-h-screen bg-gray-100">
            <div className="max-w-4xl mx-auto p-4">
              <h1 className="text-2xl font-bold">Welcome, {userName}!</h1>
              <button
                onClick={signOutUser}
                className="bg-red-500 hover:bg-red-600 text-white py-2 px-4 rounded transition"
              >
                Sign Out
              </button>

              <div className="bg-white p-6 rounded-lg shadow">
                <h2 className="text-xl font-semibold mb-4">Your Profile</h2>
                {user?.photoURL && (
                  <img
                    src={user?.photoURL}
                    alt="Profile"
                    className="w-20 h-20 rounded-full mb-4"
                  />
                )}
                <p>Email: {user?.email}</p>
              </div>
            </div>
          </div>
          <button
            onClick={toggleDarkMode}
            className={`p-2 rounded-full ${
              darkMode ? "bg-gray-700" : "bg-gray-200"
            }`}
          >
            {darkMode ? "☀️" : "🌙"}
          </button>
        </header>

        {/* Main Dashboard */}
        <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
          {/* Left Column */}
          <div className="md:col-span-2 space-y-6">
            {/* Greeting Card */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-2">
                Good Morning{name ? `, ${user.displayName}` : ""}.
              </h2>
              <p className="opacity-80">You have 0 tasks due today.</p>
            </div>

            {/* Timetable Generator */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">
                Timetable Generator
              </h2>

              <div className="mb-4">
                <label className="block font-medium mb-2">Your Name</label>
                <input
                  type="text"
                  value={name}
                  onChange={(e) => setName(e.target.value)}
                  className={`w-full p-3 rounded-lg border ${borderColor} ${inputBg}`}
                  placeholder="Enter your name"
                />
              </div>

              {isGenerating ? (
                <div className="mb-4">
                  <div className="h-2 w-full bg-gray-200 rounded-full overflow-hidden">
                    <div
                      className="h-full bg-blue-500 transition-all duration-300"
                      style={{ width: `${generationProgress}%` }}
                    ></div>
                  </div>
                  <p className="text-center mt-2">
                    Generating your timetable...{" "}
                    {Math.round(generationProgress)}%
                  </p>
                </div>
              ) : null}

              <button
                onClick={handleGenerateTimetable}
                disabled={isGenerating}
                className={`w-full py-3 px-4 rounded-lg font-medium ${
                  isGenerating
                    ? "bg-gray-400 cursor-not-allowed"
                    : "bg-blue-500 hover:bg-blue-600 text-white"
                }`}
              >
                {isGenerating ? "Generating..." : "Generate Timetable"}
              </button>
            </div>

            {/* Generated Timetable */}
            {table.length > 0 && (
              <div
                className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
              >
                <TimetablePDF data={table} name={name} darkMode={darkMode} />
              </div>
            )}

            {/* Calendar Section */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">Calendar</h2>
              <p className="opacity-70">No items found</p>
            </div>
          </div>

          {/* Right Column */}
          <div className="space-y-6">
            {/* Activities */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">Activities</h2>
              <h3 className="font-medium mb-2">Upcoming exams</h3>
              <p className="opacity-70 mb-4">No exams scheduled yet</p>
              <button className="flex items-center text-blue-500 font-medium">
                <span className="mr-1">+</span> Quick add task
              </button>
            </div>

            {/* Focus Timer */}
            <div
              className={`p-6 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <h2 className="text-xl font-semibold mb-4">Focus Timer</h2>
              <div className="mb-3">
                <label className="block font-medium mb-1">Subject</label>
                <input
                  type="text"
                  className={`w-full p-3 rounded-lg border ${borderColor} ${inputBg}`}
                  placeholder="What are you studying?"
                />
              </div>
              <div className="mb-4">
                <label className="block font-medium mb-1">Time</label>
                <input
                  type="time"
                  className={`w-full p-3 rounded-lg border ${borderColor} ${inputBg}`}
                />
              </div>
              <button
                className={`w-full py-3 px-4 rounded-lg font-medium ${
                  darkMode
                    ? "bg-green-600 hover:bg-green-700"
                    : "bg-green-500 hover:bg-green-600"
                } text-white`}
              >
                Start Focus Session
              </button>
            </div>

            {/* Date Navigation */}
            <div
              className={`p-4 rounded-xl ${cardBg} shadow border ${borderColor}`}
            >
              <div className="flex justify-between">
                {["Today", "Tomorrow", "Custom Date"].map((tab) => (
                  <button
                    key={tab}
                    onClick={() => setActiveTab(tab)}
                    className={`px-3 py-2 rounded-md font-medium ${
                      activeTab === tab
                        ? darkMode
                          ? "bg-blue-900 text-blue-300"
                          : "bg-blue-100 text-blue-600"
                        : darkMode
                        ? "hover:bg-gray-700"
                        : "hover:bg-gray-100"
                    }`}
                  >
                    {tab}
                  </button>
                ))}
              </div>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};

export default Dashboard;
